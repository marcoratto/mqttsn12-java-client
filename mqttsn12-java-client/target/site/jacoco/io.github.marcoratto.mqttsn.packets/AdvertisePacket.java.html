package io.github.marcoratto.mqttsn.packets;

import java.nio.ByteBuffer;


public class AdvertisePacket {

    private byte length;
    private byte type;
    private byte gwID;
    private short duration; 
	
	public AdvertisePacket() {
		this.length = 0;
		this.type = 0;
		this.gwID = 0;
		this.duration = 0;
	}
    
	public void decode(byte[] value) {
		ByteBuffer buffer = ByteBuffer.wrap(value);
		
		this.length = (byte) buffer.get();
		this.type = (byte) buffer.get();
		this.gwID = (byte) buffer.get();
		this.duration = buffer.getShort();
	}

	public byte getLength() {
		return length;
	}
	
	public byte getType() {
		return type;
	}
	
	public byte getGatewayID() {
		return gwID;
	}

	public short getDuration() {
		return duration;
	}

}
